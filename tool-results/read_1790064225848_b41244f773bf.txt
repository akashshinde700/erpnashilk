   101→function isTypingTarget(t: EventTarget | null): boolean {

   106→function readJsonArray(key: string): string[] {
   107→  try {
   108→    const raw = window.localStorage.getItem(key)
   109→    const parsed = raw ? JSON.parse(raw) : []
   110→    return Array.isArray(parsed) ? parsed.filter((k) => typeof k === 'string') : []
   111→  } catch { return [] }
   112→}
   113→
   114→function persistJson(key: string, value: string[]) {
   115→  try { window.localStorage.setItem(key, JSON.stringify(value)) } catch { /* private mode */ }
   116→}

   257→  function goToModule(key: string, preset?: { q?: string }) {
   258→    // Filter preset: sessionStorage covers not-yet-mounted modules, the live event covers the already-mounted one
   259→    if (preset && typeof preset.q === 'string' && preset.q.length > 0) {
   260→      try {
   261→        window.sessionStorage.setItem('erp_filter_preset', JSON.stringify({ module: key, q: preset.q, ts: Date.now() }))
   262→      } catch { /* private mode */ }
   263→      window.dispatchEvent(new CustomEvent('erp:filter-preset', { detail: { module: key, q: preset.q } }))
   264→    }
   265→    setActiveKey(key)
   266→    setPaletteOpen(false)
   267→    setPaletteQuery('') // programmatic close bypasses onOpenChange — clear stale query/results
   268→    setSearchResults([])
   269→    setNavOpen(false)
   270→    setRecents((prev) => {
   271→      const nx = [key, ...prev.filter((k) => k !== key)].slice(0, 6)
   272→      persistJson(RECENT_KEY, nx)
   273→      return nx
   274→    })
   275→  }

   281→      const detail: unknown = (e as CustomEvent).detail
   282→      if (typeof detail === 'string') {

   293→      }
   294→    }
   295→    window.addEventListener('erp:goto', handler)
   296→    return () => window.removeEventListener('erp:goto', handler)

   315→  // ---------- keyboard shortcuts engine ----------
   316→  React.useEffect(() => {
   317→    if (!session) return

   319→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
   320→        e.preventDefault()
   321→        setPaletteOpen((o) => !o)
   322→        return
   323→      }
   324→      if (paletteOpen || helpOpen || pwdOpen) return
   325→      if (isTypingTarget(e.target) || e.metaKey || e.ctrlKey || e.altKey) return
   326→
   327→      // g + key sequence (Gmail-style)
   328→      const now = Date.now()
   329→      if (goPendingRef.current && now - goPendingRef.current < 1500) {
   330→        goPendingRef.current = 0
   331→        const target = GO_KEYS[e.key.toLowerCase()]
   332→        if (target) {
   333→          e.preventDefault()
   334→          const def = findModule(target)
   335→          if (def && canAccessModule(session.user.role, target)) goToModule(target)
   336→          else if (def) toast.error(`${def.label} is not available for the ${ROLE_LABEL[session.user.role] || session.user.role} role`)
   337→          return
   338→        }
   339→      }
   340→      if (e.key === 'g') { goPendingRef.current = Date.now(); return }
   341→      if (e.key === '?') { e.preventDefault(); setHelpOpen((o) => !o); return }
   342→      if (e.key === '/') { e.preventDefault(); setPaletteOpen(true); return }
   343→      if (e.key === '[' || e.key === ']') {
   344→        e.preventDefault()
   345→        const dir = e.key === ']' ? 1 : -1
   346→        const idx = visibleModules.findIndex((m) => m.key === activeKey)
   347→        const nextIdx = (idx + dir + visibleModules.length) % visibleModules.length
   348→        const target = visibleModules[nextIdx]
   349→        if (target) goToModule(target.key)
   350→      }
   351→    }
   352→    window.addEventListener('keydown', onKey)
   353→    return () => window.removeEventListener('keydown', onKey)
   354→  }, [session, paletteOpen, helpOpen, pwdOpen, visibleModules, activeKey])
