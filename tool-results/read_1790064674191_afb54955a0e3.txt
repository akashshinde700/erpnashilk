    71→function csvEscape(v: string | number): string {
    72→  const s = String(v ?? '')
    73→  return /[",\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s
    74→}
    75→
    76→export function downloadCsv(filename: string, headers: string[], rows: (string | number)[][]) {
    77→  const csv = [headers, ...rows].map((r) => r.map(csvEscape).join(',')).join('\n')
    78→  const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8;' })
    79→  const url = URL.createObjectURL(blob)
    80→  const a = document.createElement('a')
    81→  a.href = url
    82→  a.download = `${filename}-${new Date().toISOString().slice(0, 10)}.csv`
    83→  a.click()
    84→  URL.revokeObjectURL(url)
    85→}

    89→/**
    90→ * Builds a real Excel workbook (one sheet) from a 2-D array — first data row is
    91→ * the header row. Numbers stay number cells (Excel can sum/filter them) and each
    92→ * column gets a width derived from its longest content (capped). Browser-only:
    93→ * XLSX.writeFile triggers the file download.
    94→ */
    95→export function downloadXlsx(filename: string, headers: string[], rows: (string | number)[][]) {
    96→  const ws = XLSX.utils.aoa_to_sheet([headers, ...rows])
    97→  ws['!cols'] = headers.map((h, col) => {
    98→    let max = String(h ?? '').length
    99→    for (const row of rows) {
   100→      const len = String(row[col] ?? '').length
   101→      if (len > max) max = len
   102→    }
   103→    return { wch: Math.min(Math.max(max + 2, 10), 45) }
   104→  })
   105→  const wb = XLSX.utils.book_new()
   106→  XLSX.utils.book_append_sheet(wb, ws, 'Data')
   107→  XLSX.writeFile(wb, `${filename}.xlsx`)
   108→}

   125→// ---------- Deep-link row highlight (global search → module table) ----------
   126→
   127→/** Escape a string for safe use inside a CSS attribute selector (ids are plain cuids — defensive only). */
   128→function cssAttrEscape(s: string): string {
   129→  return typeof CSS !== 'undefined' && typeof CSS.escape === 'function'
   130→    ? CSS.escape(s)
   131→    : s.replace(/["\\]/g, '\\$&')
   132→}
   133→
   134→/**
   135→ * Reads the deep-link written by the global search palette (sessionStorage key
   136→ * 'erp_deep_link': { module, id, ts }) and returns the record id that should be
   137→ * highlighted in this module's table — or null. The entry is consumed (removed)
   138→ * on first read and the highlight auto-clears after ~8s, making it a one-shot pulse.
   139→ * Pass the module's key (e.g. 'items', 'sales-orders'); deep links for other
   140→ * modules are ignored (and consumed) by the first mounted module that reads them.
   141→ */
   142→export function useRowHighlight(moduleKey: string | null | undefined): string | null {
   143→  const [highlightId, setHighlightId] = React.useState<string | null>(null)
   144→
   145→  // Read once per mount. The sessionStorage key is removed on every successful
   146→  // read, so effect re-runs (moduleKey change without a remount) are no-ops.
   147→  React.useEffect(() => {
   148→    try {
   149→      const raw = sessionStorage.getItem('erp_deep_link')
   150→      if (!raw) return
   151→      const d = JSON.parse(raw) as { module?: unknown; id?: unknown; ts?: unknown }
   152→      sessionStorage.removeItem('erp_deep_link')
   153→      if (d && d.module === moduleKey && typeof d.id === 'string' && d.id.trim() !== '') {
   154→        setHighlightId(String(d.id))
   155→      }
   156→    } catch {
   157→      // Malformed / unavailable deep link — best-effort UX nicety, ignore.
   158→    }
   159→  }, [moduleKey])
   160→
   161→  // Auto-clear the pulse so a stale highlight never sticks around.
   162→  React.useEffect(() => {
   163→    if (!highlightId) return
   164→    const t = setTimeout(() => setHighlightId(null), 8000)
   165→    return () => clearTimeout(t)
   166→  }, [highlightId])
   167→
   168→  // Same-module deep link while already mounted (no remount happens): the palette
   169→  // also broadcasts 'erp:deep-link' so we re-highlight live.
   170→  React.useEffect(() => {
   171→    const handler = (e: Event) => {
   172→      try {
   173→        const d = (e as CustomEvent<{ module?: unknown; id?: unknown }>).detail
   174→        if (d && d.module === moduleKey && typeof d.id === 'string' && d.id.trim() !== '') {
   175→          setHighlightId(null) // restart the pulse even for the same row
   176→          window.setTimeout(() => setHighlightId(String(d.id)), 0)
   177→        }
   178→      } catch {
   179→        // best-effort UX nicety — ignore
   180→      }
   181→    }
   182→    window.addEventListener('erp:deep-link', handler)
   183→    return () => window.removeEventListener('erp:deep-link', handler)
   184→  }, [moduleKey])
   185→
   186→  return highlightId
   187→}

   189→// ---------- Saved views (per-module localStorage search presets) ----------
   190→
   191→interface SavedView { name: string; q: string; ts: number }
   192→
   193→const VIEWS_STORAGE_PREFIX = 'manuflow:views:'
   194→
   195→/** Reads + sanitizes the saved-view list for a storage key. Private-mode safe (returns [] on any failure). */
   196→function loadSavedViews(key: string): SavedView[] {
   197→  try {
   198→    const raw = localStorage.getItem(VIEWS_STORAGE_PREFIX + key)
   199→    if (!raw) return []
   200→    const parsed: unknown = JSON.parse(raw)
   201→    if (!Array.isArray(parsed)) return []
   202→    return parsed
   203→      .filter((v): v is SavedView =>
   204→        !!v && typeof v === 'object' &&
   205→        typeof (v as SavedView).name === 'string' && typeof (v as SavedView).q === 'string')
   206→      .map((v) => ({ name: v.name, q: v.q, ts: typeof v.ts === 'number' ? v.ts : 0 }))
   207→  } catch {
   208→    return []
   209→  }
   210→}
   211→
   212→/** Writes the saved-view list for a storage key. Private-mode safe (silently no-ops). */
   213→function persistSavedViews(key: string, views: SavedView[]) {
   214→  try {
   215→    localStorage.setItem(VIEWS_STORAGE_PREFIX + key, JSON.stringify(views))
   216→  } catch {
   217→    // Storage unavailable (private mode / quota) — views stay in-memory for this session.
   218→  }
   219→}

   231→export function DataTable<T extends { id: string }>({
   232→  columns, rows, loading, emptyText = 'No records yet', searchPlaceholder, onRowClick, maxHeight = 'auto', pageSize = 12, csvExport, xlsxExport, highlightId, moduleKey, savedViewsKey, serverPagination,
   233→}: {
   234→  columns: Column<T>[]
   235→  rows: T[]
   236→  loading?: boolean
   237→  emptyText?: string
   238→  searchPlaceholder?: string
   239→  onRowClick?: (row: T) => void
   240→  maxHeight?: string
   241→  pageSize?: number
   242→  /** When provided, shows an "Export CSV" button exporting all filtered rows. */
   243→  csvExport?: { filename: string; pick: (row: T) => (string | number)[] }
   244→  /** When provided, shows an "Excel" button (beside the CSV one) exporting the same filtered rows as a real .xlsx. Headers default to the column labels — identical columns to the CSV. */
   245→  xlsxExport?: { filename: string; headers?: string[]; pick: (row: T) => (string | number)[] }
   246→  /** Deep-linked record id (from useRowHighlight) — page-jumps to + scrolls the row into view and highlights it. */
   247→  highlightId?: string | null
   248→  /** Opts into the dashboard filter-preset channel: on mount consumes sessionStorage 'erp_filter_preset' ({ module, q, ts }) when its module matches, and live-listens for 'erp:filter-preset' CustomEvents (detail { module, q }). Pass the module's key (e.g. 'items'). No effect when omitted. */
   249→  moduleKey?: string
   250→  /** Enables the "Views" saved-search control. Views persist in localStorage under `manuflow:views:<savedViewsKey>` — pass `${companyId}:<moduleKey>` so views are scoped per company + module. No effect when omitted. */
   251→  savedViewsKey?: string
   252→  /** Opts into server-side pagination: `rows` must then be exactly the page slice the API returned (already filtered + paginated). The local text filter and internal page state are bypassed — Prev/Next and the "Showing …" footer are driven by `page`/`onPageChange`/`total`, and (with `onSearchChange`) the search text is pushed to the module debounced by 300ms (mount run skipped). Omit for the default fully client-side behavior (zero change). */
   253→  serverPagination?: {
   254→    /** 0-based page index, controlled by the module (Prev → onPageChange(page - 1), Next → onPageChange(page + 1)). */
   255→    page: number
   256→    /** Rows per server page — the module must fetch exactly this many. */
   257→    pageSize: number
   258→    /** Full filtered row count from the server (across all pages). */
   259→    total: number
   260→    /** Called when the user pages (Prev/Next) or a highlight page-jump targets another page. */
   261→    onPageChange: (page: number) => void
   262→    /** Debounced (300ms) search-text push; the module should store it as its server-side q filter and reset to page 0. */
   263→    onSearchChange?: (q: string) => void
   264→  }
   265→}) {
   266→  const [q, setQ] = React.useState('')
   267→  const [appliedView, setAppliedView] = React.useState<string | null>(null)
   268→  const [page, setPage] = React.useState(0)
   269→  const wrapperRef = React.useRef<HTMLDivElement | null>(null)
   270→
   271→  const filtered = React.useMemo(() => {
   272→    if (serverPagination) return rows // server mode: the API already filtered + sliced — rows are the page as-is
   273→    if (!q.trim()) return rows
   274→    const needle = q.toLowerCase()
   275→    return rows.filter((r) => JSON.stringify(r).toLowerCase().includes(needle))
   276→  }, [rows, q, serverPagination])
   277→
   278→  React.useEffect(() => setPage(0), [q, rows.length])
   279→
   280→  // Deep-link: page-jump to + scroll the highlighted row into view. Rows may
   281→  // arrive async (table can be loading when the highlight first applies), so
   282→  // attempts are retried across a few ticks; state kept in a ref keyed by the
   283→  // highlight id makes each jump a strict one-shot that never fights the
   284→  // user's own paging/filtering afterwards.
   285→  const jumpStateRef = React.useRef<{ id: string | null; jumped: boolean; scrolled: boolean }>({ id: null, jumped: false, scrolled: false })
   286→
   287→  // Latest serverPagination config, readable inside effects below without
   288→  // re-triggering them on every render (the object is recreated per render).
   289→  // Synced in a bare effect (runs after every render, before the consumers
   290→  // declared after it) so effects always read the freshest config.
   291→  const serverPaginationRef = React.useRef(serverPagination)
   292→  React.useEffect(() => {
   293→    serverPaginationRef.current = serverPagination
   294→  })
   295→
   296→  React.useEffect(() => {
   297→    if (!highlightId) return
   298→    const st = jumpStateRef.current
   299→    if (st.id !== highlightId) {
   300→      st.id = highlightId
   301→      st.jumped = false
   302→      st.scrolled = false
   303→    }
   304→    const timers: ReturnType<typeof setTimeout>[] = []
   305→    const attempt = () => {
   306→      if (!st.jumped) {
   307→        const idx = filtered.findIndex((r) => String(r.id) === highlightId)
   308→        if (idx === -1) return // not in filtered yet (rows still loading) or filtered out by the text filter — do nothing
   309→        st.jumped = true
   310→        const sp = serverPaginationRef.current
   311→        if (sp) sp.onPageChange(Math.floor(idx / sp.pageSize))
   312→        else setPage(Math.floor(idx / pageSize))
   313→      }
   314→      if (!st.scrolled) {
   315→        const el = wrapperRef.current?.querySelector(`[data-row-id="${cssAttrEscape(highlightId)}"]`)
   316→        if (el) {
   317→          el.scrollIntoView({ block: 'center', behavior: 'smooth' })
   318→          st.scrolled = true
   319→        }
   320→      }
   321→    }
   322→    for (const delay of [0, 150, 400, 900]) timers.push(setTimeout(attempt, delay))
   323→    return () => timers.forEach(clearTimeout)
   324→  }, [highlightId, filtered, pageSize])
   325→
   326→  // ---- Dashboard filter presets (moduleKey) ----
   327→  // One-shot: read + consume sessionStorage 'erp_filter_preset' ({ module, q, ts })
   328→  // on mount, exactly like the deep-link pattern above.
   329→  React.useEffect(() => {
   330→    if (!moduleKey) return
   331→    try {
   332→      const raw = sessionStorage.getItem('erp_filter_preset')
   333→      if (!raw) return
   334→      const d = JSON.parse(raw) as { module?: unknown; q?: unknown; ts?: unknown }
   335→      sessionStorage.removeItem('erp_filter_preset')
   336→      if (d && d.module === moduleKey && typeof d.q === 'string') setQ(d.q)
   337→    } catch {
   338→      // Malformed / unavailable preset — best-effort, ignore.
   339→    }
   340→  }, [moduleKey])
   341→
   342→  // Same-module live presets while already mounted (no remount happens): the
   343→  // emitter also broadcasts 'erp:filter-preset' with detail { module, q }.
   344→  React.useEffect(() => {
   345→    if (!moduleKey) return
   346→    const handler = (e: Event) => {
   347→      try {
   348→        const d = (e as CustomEvent<{ module?: unknown; q?: unknown }>).detail
   349→        if (d && d.module === moduleKey && typeof d.q === 'string') {
   350→          const next = String(d.q)
   351→          setQ((prev) => (next === prev ? prev : next))
   352→          setAppliedView(null) // an external preset replaces any applied view
   353→        }
   354→      } catch {
   355→        // best-effort — ignore
   356→      }
   357→    }
   358→    window.addEventListener('erp:filter-preset', handler)
   359→    return () => window.removeEventListener('erp:filter-preset', handler)
   360→  }, [moduleKey])
   361→
   362→  // ---- Saved views (savedViewsKey) ----
   363→  const [views, setViews] = React.useState<SavedView[]>([])
   364→  const [saveDialogOpen, setSaveDialogOpen] = React.useState(false)
   365→  const [viewName, setViewName] = React.useState('')
   366→
   367→  // Load (re-load on key change) from localStorage after mount — hydration safe.
   368→  React.useEffect(() => {
   369→    if (!savedViewsKey) return
   370→    setViews(loadSavedViews(savedViewsKey))
   371→    setAppliedView(null)
   372→  }, [savedViewsKey])
   373→
   374→  const applyView = (v: SavedView) => {
   375→    setQ(v.q)
   376→    setAppliedView(v.name)
   377→  }
   378→
   379→  const deleteView = (name: string) => {
   380→    if (!savedViewsKey) return
   381→    const next = views.filter((v) => v.name !== name)
   382→    setViews(next)
   383→    persistSavedViews(savedViewsKey, next)
   384→    setAppliedView((cur) => (cur === name ? null : cur))
   385→  }
   386→
   387→  const saveCurrentView = () => {
   388→    if (!savedViewsKey) return
   389→    const name = viewName.trim() || 'All records'
   390→    const view: SavedView = { name, q, ts: Date.now() }
   391→    const idx = views.findIndex((v) => v.name === name)
   392→    const next = idx === -1 ? [...views, view] : views.map((v, i) => (i === idx ? view : v))
   393→    setViews(next)
   394→    persistSavedViews(savedViewsKey, next)
   395→    setAppliedView(name)
   396→    setSaveDialogOpen(false)
   397→  }
   398→
   399→  // ---- Server-side search (serverPagination.onSearchChange) ----
   400→  // The search input still owns the local `q` (saved views + presets keep
   401→  // working visually); when an onSearchChange callback is wired, q is pushed
   402→  // to the module debounced by 300ms. The very first run after mount is
   403→  // skipped, and values already sent are suppressed — so programmatic setQ
   404→  // (views/presets) to the same text never triggers a redundant refetch.
   405→  const onSearchChangeRef = React.useRef(serverPagination?.onSearchChange)
   406→  React.useEffect(() => {
   407→    onSearchChangeRef.current = serverPagination?.onSearchChange
   408→  })
   409→  const serverSearchArmedRef = React.useRef(false)
   410→  const serverSearchLastSentRef = React.useRef<string | null>(null)
   411→
   412→  React.useEffect(() => {
   413→    const cb = onSearchChangeRef.current
   414→    if (!cb) return
   415→    if (!serverSearchArmedRef.current) {
   416→      serverSearchArmedRef.current = true
   417→      serverSearchLastSentRef.current = q
   418→      return
   419→    }
   420→    if (q === serverSearchLastSentRef.current) return
   421→    const timer = setTimeout(() => {
   422→      serverSearchLastSentRef.current = q
   423→      cb(q)
   424→    }, 300)
   425→    return () => clearTimeout(timer) // re-armed per q change; cleared on unmount
   426→  }, [q])
   427→
   428→  const totalPages = serverPagination
   429→    ? Math.max(1, Math.ceil(serverPagination.total / serverPagination.pageSize))
   430→    : Math.max(1, Math.ceil(filtered.length / pageSize))
   431→  const pageRows = serverPagination ? filtered : filtered.slice(page * pageSize, page * pageSize + pageSize)
   432→  const pagerPage = serverPagination ? serverPagination.page : page
   433→  const pagerPageSize = serverPagination ? serverPagination.pageSize : pageSize
   434→  const pagerTotal = serverPagination ? serverPagination.total : filtered.length
   435→  const gotoPage = (p: number) => {
   436→    if (serverPagination) serverPagination.onPageChange(p)
   437→    else setPage(p)
   438→  }
   439→
   440→  return (
   441→    <div className="space-y-3">
   442→      {(searchPlaceholder || csvExport || xlsxExport || savedViewsKey) && (
   443→        <div className="flex items-center justify-between gap-2">
   444→          {searchPlaceholder ? (
   445→            <div className="relative max-w-xs flex-1">
   446→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   447→              <Input value={q} onChange={(e) => { setQ(e.target.value); setAppliedView(null) }} placeholder={searchPlaceholder} className="pl-8 h-9" />
   448→            </div>
   449→          ) : <div />}
   450→          <div className="flex items-center gap-2 shrink-0">
   451→            {csvExport && rows.length > 0 && (
   452→              <Button
   453→                variant="outline"
   454→                size="sm"
   455→                className="h-9"
   456→                onClick={() => downloadCsv(csvExport.filename, columns.map((c) => c.header), filtered.map(csvExport.pick))}
   457→              >
   458→                <Download className="h-4 w-4 mr-1.5" />CSV
   459→              </Button>
   460→            )}
   461→            {xlsxExport && rows.length > 0 && (
   462→              <Button
   463→                variant="outline"
   464→                size="sm"

   473→                {appliedView && (
   474→                  <span
   475→                    className="inline-flex items-center gap-1 max-w-40 rounded-full bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300 px-2 py-0.5 text-[11px] font-medium"
   476→                    title={`Applied view: ${appliedView}`}
   477→                  >
   478→                    <span className="truncate">{appliedView}</span>
   479→                    <button
   480→                      type="button"
   481→                      title="Clear applied view"
   482→                      className="rounded-full p-0.5 hover:bg-amber-200/70 dark:hover:bg-amber-900/60"
   483→                      onClick={() => { setQ(''); setAppliedView(null) }}
   484→                    >
   485→                      <X className="h-3 w-3" />
   486→                    </button>
   487→                  </span>
   488→                )}
   489→                <DropdownMenu>
   490→                  <DropdownMenuTrigger asChild>
   491→                    <Button variant="outline" size="sm" className="h-9" title="Saved views">

   493→                      <span className="hidden sm:inline">Views</span>
   494→                    </Button>
   495→                  </DropdownMenuTrigger>
   496→                  <DropdownMenuContent align="end" className="w-64">
   497→                    {views.length === 0 ? (
   498→                      <div className="px-2 py-1.5 text-xs text-muted-foreground">No saved views yet</div>
   499→                    ) : (
   500→                      views.map((v) => (
   501→                        <DropdownMenuItem key={v.name} onSelect={() => applyView(v)} className="items-start gap-2">
   502→                          <div className="min-w-0 flex-1 py-0.5">
   503→                            <div className="font-medium truncate">{v.name}</div>

   505→                          </div>
   506→                          <button
   507→                            type="button"
   508→                            title={`Delete view “${v.name}”`}

   510→                            onClick={(e) => { e.stopPropagation(); e.preventDefault(); deleteView(v.name) }}

   517→                    <DropdownMenuSeparator />
   518→                    <DropdownMenuItem
   519→                      onSelect={() => { setViewName(q.trim() || 'All records'); setSaveDialogOpen(true) }}
   520→                    >
   521→                      <BookmarkPlus className="h-4 w-4 mr-2" />
   522→                      Save current search…
   523→                    </DropdownMenuItem>
   524→                  </DropdownMenuContent>
   525→                </DropdownMenu>
   526→              </div>
   527→            )}
   528→          </div>
   529→        </div>
   530→      )}
   531→      <div ref={wrapperRef} className="rounded-lg border overflow-hidden bg-card">
   532→        <ScrollArea className={maxHeight !== 'auto' ? maxHeight : undefined} style={maxHeight !== 'auto' ? { height: maxHeight } : undefined}>
   533→          <Table>
   534→            <TableHeader>
   535→              <TableRow className="bg-muted/50 hover:bg-muted/50">
   536→                {columns.map((c) => (
   537→                  <TableHead key={c.key} className={cn('font-medium whitespace-nowrap', c.hideOnMobile && 'hidden md:table-cell', c.className)}>{c.header}</TableHead>
   538→                ))}
   539→              </TableRow>
   540→            </TableHeader>
   541→            <TableBody>
   542→              {loading ? (
   543→                Array.from({ length: 5 }).map((_, i) => (
   544→                  <TableRow key={`sk-${i}`}>
   545→                    {columns.map((c) => (
   546→                      <TableCell key={c.key} className={cn(c.hideOnMobile && 'hidden md:table-cell')}><Skeleton className="h-4 w-full max-w-24" /></TableCell>
   547→                    ))}
   548→                  </TableRow>
   549→                ))
   550→              ) : pageRows.length === 0 ? (
   551→                <TableRow>
   552→                  <TableCell colSpan={columns.length} className="h-40 text-center">
   553→                    <div className="flex flex-col items-center gap-2 text-muted-foreground">
   554→                      <Inbox className="h-8 w-8" />
   555→                      <p className="text-sm">{emptyText}</p>
   556→                    </div>
   557→                  </TableCell>
   558→                </TableRow>
   559→              ) : (
   560→                pageRows.map((row) => {
   561→                  const highlighted = highlightId !== null && highlightId !== undefined && String(row.id) === highlightId
   562→                  return (
   563→                    <TableRow
   564→                      key={row.id}
   565→                      data-row-id={String(row.id)}
   566→                      onClick={() => onRowClick?.(row)}
   567→                      className={cn(
   568→                        onRowClick && 'cursor-pointer',
   569→                        highlighted && 'bg-amber-50 dark:bg-amber-500/10 ring-2 ring-inset ring-amber-300 dark:ring-amber-500/40',
   570→                      )}
   571→                    >
   572→                      {columns.map((c) => (
   573→                        <TableCell key={c.key} className={cn('align-middle', c.hideOnMobile && 'hidden md:table-cell', c.className)}>{c.render(row)}</TableCell>
   574→                      ))}
   575→                    </TableRow>
   576→                  )
   577→                })
   578→              )}
   579→            </TableBody>
   580→          </Table>
   581→        </ScrollArea>
   582→      </div>
   583→      {pagerTotal > pagerPageSize && (
   584→        <div className="flex items-center justify-between text-sm text-muted-foreground">
   585→          <span>Showing {pagerPage * pagerPageSize + 1}–{Math.min((pagerPage + 1) * pagerPageSize, pagerTotal)} of {pagerTotal}</span>
   586→          <div className="flex gap-2">
   587→            <Button variant="outline" size="sm" disabled={pagerPage === 0} onClick={() => gotoPage(pagerPage - 1)}>Prev</Button>
   588→            <Button variant="outline" size="sm" disabled={pagerPage >= totalPages - 1} onClick={() => gotoPage(pagerPage + 1)}>Next</Button>
   589→          </div>
   590→        </div>
   591→      )}
   592→      {savedViewsKey && (
   593→        <Dialog open={saveDialogOpen} onOpenChange={setSaveDialogOpen}>
   594→          <DialogContent className="sm:max-w-sm">
   595→            <DialogHeader>
   596→              <DialogTitle>Save current search</DialogTitle>
   597→              <DialogDescription>
   598→                {q.trim() !== ''
   599→                  ? <>Saves the current search text: <span className="font-medium">“{q}”</span></>
   600→                  : 'Saves an empty search (shows all records)'}
   601→              </DialogDescription>
   602→            </DialogHeader>
   603→            <form onSubmit={(e) => { e.preventDefault(); saveCurrentView() }}>
   604→              <div className="grid gap-4 py-2">
   605→                <Field label="View name" required>
   606→                  <Input value={viewName} onChange={(e) => setViewName(e.target.value)} placeholder="All records" autoFocus />
   607→                </Field>
   608→              </div>
   609→              <DialogFooter className="mt-2">
   610→                <Button type="button" variant="outline" onClick={() => setSaveDialogOpen(false)}>Cancel</Button>
   611→                <Button type="submit">Save</Button>
   612→              </DialogFooter>
   613→            </form>
   614→          </DialogContent>
   615→        </Dialog>
   616→      )}
   617→    </div>
   618→  )
   619→}

   687→export function SelectInput({ value, onChange, options, placeholder = 'Select…', disabled }: {
   688→  value: string; onChange: (v: string) => void; options: { value: string; label: string }[]; placeholder?: string; disabled?: boolean
   689→}) {
   690→  // Radix Select forbids empty string item values — map '' to a sentinel transparently
   691→  const EMPTY = '__EMPTY__'
   692→  const safeOptions = options.map((o) => (o.value === '' ? { ...o, value: EMPTY } : o))
   693→  const safeValue = value === '' ? EMPTY : (value || undefined)
   694→  return (
   695→    <Select value={safeValue} onValueChange={(v) => onChange(v === EMPTY ? '' : v)} disabled={disabled}>
   696→      <SelectTrigger className="h-9 w-full"><SelectValue placeholder={placeholder} /></SelectTrigger>
   697→      <SelectContent className="max-h-72">
   698→        {safeOptions.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   699→      </SelectContent>
   700→    </Select>
   701→  )
   702→}
