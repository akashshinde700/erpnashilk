   101→function isTypingTarget(t: EventTarget | null): boolean {
   102→  if (!(t instanceof HTMLElement)) return false

   106→function readJsonArray(key: string): string[] {
   107→  try {
   108→    const raw = window.localStorage.getItem(key)
   109→    const parsed = raw ? JSON.parse(raw) : []
   110→    return Array.isArray(parsed) ? parsed.filter((k) => typeof k === 'string') : []
   111→  } catch { return [] }
   112→}
   113→
   114→function persistJson(key: string, value: string[]) {
   115→  try { window.localStorage.setItem(key, JSON.stringify(value)) } catch { /* private mode */ }
   116→}

   301→        e.preventDefault()

   303→        return
   304→      }

   306→      if (isTypingTarget(e.target) || e.metaKey || e.ctrlKey || e.altKey) return
   307→

   309→      const now = Date.now()
   310→      if (goPendingRef.current && now - goPendingRef.current < 1500) {
   311→        goPendingRef.current = 0
   312→        const target = GO_KEYS[e.key.toLowerCase()]
   313→        if (target) {
   314→          e.preventDefault()
   315→          const def = findModule(target)
   316→          if (def && canAccessModule(session.user.role, target)) goToModule(target)
   317→          else if (def) toast.error(`${def.label} is not available for the ${ROLE_LABEL[session.user.role] || session.user.role} role`)
   318→          return
   319→        }
   320→      }
   321→      if (e.key === 'g') { goPendingRef.current = Date.now(); return }
   322→      if (e.key === '?') { e.preventDefault(); setHelpOpen((o) => !o); return }
   323→      if (e.key === '/') { e.preventDefault(); setPaletteOpen(true); return }
   324→      if (e.key === '[' || e.key === ']') {
   325→        e.preventDefault()
   326→        const dir = e.key === ']' ? 1 : -1
   327→        const idx = visibleModules.findIndex((m) => m.key === activeKey)
   328→        const nextIdx = (idx + dir + visibleModules.length) % visibleModules.length
   329→        const target = visibleModules[nextIdx]
   330→        if (target) goToModule(target.key)
   331→      }
   332→    }
   333→    window.addEventListener('keydown', onKey)
   334→    return () => window.removeEventListener('keydown', onKey)
   335→  }, [session, paletteOpen, helpOpen, pwdOpen, visibleModules, activeKey])

   416→  function renderModuleRow(m: (typeof MODULES)[number], opts: { collapsed: boolean; star?: boolean }) {

   426→      >
   427→        <button
   428→          onClick={() => goToModule(m.key)}
   429→          title={opts.collapsed ? m.label : undefined}
   430→          className={cn(
   431→            'flex flex-1 items-center gap-2.5 py-1.5 text-[13px] text-left min-w-0',
   432→            opts.collapsed ? 'justify-center px-0' : 'px-2.5',

   441→            onClick={() => toggleFav(m.key)}
   442→            className={cn(
   443→              'mr-1.5 rounded p-1 transition-opacity',

   445→              active ? 'hover:bg-white/20' : 'hover:bg-muted',
   446→            )}
   447→            aria-label={isFav ? `Unpin ${m.label}` : `Pin ${m.label} to favorites`}
   448→            title={isFav ? 'Unpin' : 'Pin to favorites'}
   449→          >

   451→          </button>
   452→        )}
   453→      </div>
   454→    )
   455→  }
