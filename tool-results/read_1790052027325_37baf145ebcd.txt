    33→function computeTotals(lines: DraftLine[]) {
    34→  let subtotal = 0, discount = 0, gst = 0
    35→  for (const l of lines) {
    36→    const gross = l.qty * l.rate
    37→    const disc = gross * ((l.discountPct || 0) / 100)
    38→    const net = gross - disc
    39→    subtotal += gross; discount += disc; gst += net * ((l.gstRate || 0) / 100)
    40→  }
    41→  return { subtotal: r2(subtotal), discount: r2(discount), cgst: r2(gst / 2), sgst: r2(gst / 2), grand: r2(subtotal - discount + gst) }
    42→}
    43→function emptyLine(): DraftLine { return { itemId: '', qty: 1, rate: 0, discountPct: 0, gstRate: 18 } }

    45→// Escape a value for interpolation into the print-window HTML.
    46→const esc = (v?: string | null) =>

   321→    { key: 'status', header: 'Status', render: (r) => <StatusBadge status={r.status} /> },
   322→    {
   323→      key: 'actions', header: 'Actions',
   324→      render: (r) => (
   325→        <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
   326→          <Button variant="ghost" size="sm" className="h-7" title="Print tax invoice" onClick={() => void printInvoice(r)}><Printer className="h-3.5 w-3.5" /></Button>
   327→          {!r.irn && r.status !== 'CANCELLED' && (
   328→            <Button variant="ghost" size="sm" className="h-7" title="Generate IRN" onClick={() => void generateIrn(r)}><BadgeIndianRupee className="h-3.5 w-3.5" /></Button>
   329→          )}
   330→          {r.status !== 'PAID' && r.status !== 'CANCELLED' && (
   331→            <Button variant="outline" size="sm" className="h-7" onClick={() => void openPay(r)}>Payment</Button>
   332→          )}
   333→          {r.status !== 'CANCELLED' && (r.paidAmount || 0) <= 0 && (
   334→            <ConfirmAction
   335→              trigger={<Button variant="ghost" size="sm" className="h-7 text-red-600"><X className="h-3.5 w-3.5" /></Button>}
   336→              title="Cancel invoice?" message={`Cancel ${r.number}? Only unpaid invoices can be cancelled.`} tone="destructive"
   337→              onConfirm={() => cancelInvoice(r)}
   338→            />
   339→          )}
   340→        </div>
   341→      ),
   342→    },
   343→  ]
   344→
   345→  return (
   346→    <div>
   347→      <PageHeader
   348→        title="Sales Invoices"
   349→        description="GST tax invoices, payments received and e-invoicing"
   350→        actions={<Button onClick={() => { resetForm(); setOpen(true) }}><Plus className="h-4 w-4 mr-1.5" />New Invoice</Button>}
   351→      />
   352→      <DataTable columns={columns} rows={rows} loading={list.loading} searchPlaceholder="Search invoices…" highlightId={highlightId} />
   353→
   354→      <FormDialog
   355→        open={open} onOpenChange={setOpen} wide title="New Sales Invoice" submitLabel="Create Invoice"
   356→        description="From a sales order (pending qty pre-filled) or manual lines" loading={saving} onSubmit={submit}
   357→      >
   358→        <div className="grid sm:grid-cols-3 gap-4">
   359→          <Field label="Mode">
   360→            <SelectInput
   361→              value={mode} onChange={setMode}
   362→              options={[{ value: 'SO', label: 'From Sales Order' }, { value: 'MANUAL', label: 'Manual Lines' }]}
   363→            />
   364→          </Field>
   365→          {mode === 'SO' ? (
   366→            <Field label="Sales Order" required>
   367→              <SelectInput value={salesOrderId} onChange={(v) => void pickSo(v)} options={soOpts} placeholder="Select SO…" />
   368→            </Field>
   369→          ) : (
   370→            <Field label="Customer" required>
   371→              <SelectInput value={customerId} onChange={setCustomerId} options={customerOpts} />
   372→            </Field>
   373→          )}
   374→          <Field label="Date" required><DateInput value={date} onChange={setDate} /></Field>
   375→          <Field label="Due Date"><DateInput value={dueDate} onChange={setDueDate} /></Field>
   376→          <Field label="Place of Supply"><TextInput value={placeOfSupply} onChange={(e) => setPlaceOfSupply(e.target.value)} placeholder="e.g. Maharashtra" /></Field>
   377→          <Field label="Notes"><TextInput value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Remarks…" /></Field>
   378→        </div>
   379→
   380→        <div className="space-y-2">
   381→          <div className="flex items-center justify-between">
   382→            <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Lines</span>
   383→            <Button type="button" variant="outline" size="sm" onClick={() => setLines((ls) => [...ls, emptyLine()])}>
   384→              <Plus className="h-3.5 w-3.5 mr-1" />Add Line
   385→            </Button>
   386→          </div>
   387→          {lines.map((l, idx) => (
   388→            <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   389→              <div className="col-span-12 sm:col-span-4">
   390→                <SelectInput
   391→                  value={l.itemId} options={itemOpts} placeholder="Select item…"
   392→                  onChange={(v) => setLines((ls) => ls.map((x, i) => {
   393→                    if (i !== idx) return x
   394→                    const it = (items.data?.items ?? []).find((o) => o.id === v)
   395→                    return { ...x, itemId: v, rate: x.rate || Number(it?.sellingPrice || 0), gstRate: it?.gstRate ?? x.gstRate }
   396→                  }))}
   397→                />
   398→              </div>
   399→              <NumberInput className="col-span-3 sm:col-span-2" value={l.qty} min={0} placeholder="Qty"
   400→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, qty: n } : x)))} />
   401→              <NumberInput className="col-span-3 sm:col-span-2" value={l.rate} min={0} placeholder="Rate"
   402→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, rate: n } : x)))} />
   403→              <NumberInput className="col-span-3 sm:col-span-1" value={l.discountPct} min={0} placeholder="Disc %"
   404→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, discountPct: n } : x)))} />
   405→              <NumberInput className="col-span-2 sm:col-span-1" value={l.gstRate} min={0} placeholder="GST %"
   406→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, gstRate: n } : x)))} />
   407→              <div className="col-span-1 flex justify-end">
   408→                <Button type="button" variant="ghost" size="sm" disabled={lines.length <= 1}
   409→                  onClick={() => setLines((ls) => ls.filter((_, i) => i !== idx))}>
   410→                  <Trash2 className="h-3.5 w-3.5 text-red-600" />
   411→                </Button>
   412→              </div>
   413→            </div>
   414→          ))}
   415→          <div className="rounded-md border bg-muted/30 p-3 text-sm grid grid-cols-2 sm:grid-cols-5 gap-2">
   416→            <TotCell label="Subtotal" value={totals.subtotal} />
   417→            <TotCell label="Discount" value={totals.discount} />
   418→            <TotCell label="CGST (preview)" value={totals.cgst} />
   419→            <TotCell label="SGST (preview)" value={totals.sgst} />
   420→            <TotCell label="Grand Total" value={totals.grand} bold />
   421→          </div>
   422→        </div>
   423→      </FormDialog>
   424→
   425→      <FormDialog
   426→        open={!!payFor} onOpenChange={(o) => !o && setPayFor(null)} title={`Record Payment — ${payFor?.number || ''}`}
   427→        submitLabel="Save Payment" loading={paying} onSubmit={submitPayment}
   428→        description={`Outstanding: ${fmtMoney(payFor ? (payFor.grandTotal || 0) - (payFor.paidAmount || 0) : 0)}`}
   429→      >
   430→        <div className="grid sm:grid-cols-2 gap-4">
   431→          <Field label="Amount" required><NumberInput value={payAmount} onChange={setPayAmount} min={0} /></Field>
   432→          <Field label="Mode">
   433→            <SelectInput value={payMode} onChange={setPayMode} options={PAYMENT_MODES.map((m) => ({ value: m, label: m }))} />
   434→          </Field>
   435→          <Field label="Date" required><DateInput value={payDate} onChange={setPayDate} /></Field>
   436→          <Field label="Reference"><TextInput value={payRef} onChange={(e) => setPayRef(e.target.value)} placeholder="UTR / cheque no." /></Field>
   437→        </div>
   438→      </FormDialog>
   439→    </div>
   440→  )
   441→}
   442→
   443→function TotCell({ label, value, bold }: { label: string; value: number; bold?: boolean }) {
   444→  return (
   445→    <div className={bold ? 'sm:text-right' : ''}>
   446→      <p className="text-[11px] text-muted-foreground uppercase tracking-wide">{label}</p>
   447→      <p className={`tabular-nums ${bold ? 'font-bold text-base' : 'font-medium'}`}>{fmtMoney(value)}</p>
   448→    </div>
   449→  )
   450→}
