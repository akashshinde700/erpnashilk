    33→function computeTotals(lines: DraftLine[]) {
    34→  let subtotal = 0, discount = 0, gst = 0
    35→  for (const l of lines) {
    36→    const gross = l.qty * l.rate
    37→    const disc = gross * ((l.discountPct || 0) / 100)
    38→    const net = gross - disc
    39→    subtotal += gross; discount += disc; gst += net * ((l.gstRate || 0) / 100)
    40→  }
    41→  return { subtotal: r2(subtotal), discount: r2(discount), cgst: r2(gst / 2), sgst: r2(gst / 2), grand: r2(subtotal - discount + gst) }
    42→}
    43→function emptyLine(): DraftLine { return { itemId: '', qty: 1, rate: 0, discountPct: 0, gstRate: 18 } }

    45→// Escape a value for interpolation into the print-window HTML.
    46→const esc = (v?: string | null) =>
    47→  (v ?? '').replace(/[&<>"]/g, (c) => (c === '&' ? '&amp;' : c === '<' ? '&lt;' : c === '>' ? '&gt;' : '&quot;'))
    48→// Join non-empty address parts with ', ' (raw — wrap the result in esc() before injecting).
    49→const joinAddr = (...parts: (string | null | undefined)[]) =>
    50→  parts.filter((p): p is string => !!p && p.trim() !== '').join(', ')

    52→// Build a wa.me deep link: 10-digit numbers get the '91' prefix, already-prefixed
    53→// 12-digit numbers pass through, anything else (or no phone) opens WhatsApp's chat chooser.
    54→function waUrl(phone: string | null | undefined, message: string): string {
    55→  const digits = (phone || '').replace(/\D/g, '')
    56→  const valid = digits.length === 10 || (digits.length === 12 && digits.startsWith('91'))
    57→  return `https://wa.me/${valid ? digits : ''}?text=${encodeURIComponent(message)}`
    58→}
    59→
    60→// Build a mailto: deep link — a missing recipient opens the user's mail app with a blank To: field.
    61→function mailtoUrl(email: string | null | undefined, subject: string, body: string): string {
    62→  return `mailto:${email || ''}?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`
    63→}

    66→function invoiceMessage(row: InvoiceRow, company?: CompanyRow): string {
    67→  const customerName = row.customer?.name || 'Customer'
    68→  return `Namaste ${customerName},\n\nTax Invoice ${row.number} dated ${fmtDate(row.date)} — Amount ${fmtMoney(row.grandTotal)}.\n${row.paidAmount < row.grandTotal ? `Balance due: ${fmtMoney(r2(row.grandTotal - row.paidAmount))}${row.dueDate ? ` (due ${fmtDate(row.dueDate)})` : ''}\n` : ''}${company?.bankUpiId ? `Pay via UPI: ${company.bankUpiId}\n` : ''}Thank you,\n${company?.name || 'Accounts'}`
    69→}

   323→              <tr><td>Paid</td><td class="num">${fmtMoney(inv.paidAmount)}</td></tr>
   324→              <tr><td>Balance Due</td><td class="num">${fmtMoney(balance)}</td></tr>
   325→            </table>
   326→            <div style="margin-top:6px;font-size:13px"><strong>Total Invoice Value: ${fmtMoney(inv.grandTotal)}</strong></div>
   327→            <div class="words">Amount in words: Rupees ${amountInWords(inv.grandTotal)} Only</div>
   328→          </td>
   329→        </tr></table>
   330→        ${inv.notes ? `<div class="muted" style="padding:6px 10px;border-top:1px solid #444">Notes: ${esc(inv.notes)}</div>` : ''}
   331→        <table class="bare"><tr>
   332→          <td style="width:58%">
   333→            <div class="sec">Terms &amp; Conditions</div>
   334→            <div class="terms">
   335→              1. Subject to ${seller?.city ? esc(seller.city) + ' ' : ''}jurisdiction.<br/>
   336→              2. Interest @18% p.a. will be charged on overdue payments.<br/>
   337→              3. Delivery subject to realisation of payment.<br/>
   338→              4. Goods once sold will not be taken back. E&amp;OE.
   339→            </div>
   340→          </td>
   341→          <td style="width:42%;text-align:right">
   342→            <div style="font-weight:700;font-size:12px;text-align:left">For ${esc(sellerName)}</div>
   343→            <div class="signline">Authorised Signatory</div>
   344→          </td>
   345→        </tr></table>
   346→        <div class="footer">
   347→          This is a computer-generated GST tax invoice.<br/>
   348→          <strong>Declaration:</strong> We declare that this invoice shows the actual price of the goods described and that all particulars are true and correct.
   349→        </div>
   350→      </div>`
   351→    printHtml(`Tax Invoice ${row.number}`, html)
   352→  }
   353→
   354→  const totals = computeTotals(lines)
   355→
   356→  const columns: Column<InvoiceRow>[] = [
   357→    { key: 'number', header: 'Invoice #', render: (r) => <span className="font-medium">{r.number}</span> },
   358→    { key: 'date', header: 'Date', render: (r) => <DateCell value={r.date} /> },
   359→    { key: 'customer', header: 'Customer', render: (r) => r.customer?.name || '—' },
   360→    { key: 'total', header: 'Grand Total', render: (r) => <Money value={r.grandTotal} className="font-medium" /> },
   361→    { key: 'paid', header: 'Paid', hideOnMobile: true, render: (r) => <Money value={r.paidAmount} /> },
   362→    { key: 'outstanding', header: 'Outstanding', render: (r) => <Money value={r2((r.grandTotal || 0) - (r.paidAmount || 0))} className={r2((r.grandTotal || 0) - (r.paidAmount || 0)) > 0 ? 'text-amber-600 font-medium' : 'text-emerald-600'} /> },
   363→    { key: 'status', header: 'Status', render: (r) => <StatusBadge status={r.status} /> },
   364→    {
   365→      key: 'actions', header: 'Actions',
   366→      render: (r) => (
   367→        <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
   368→          <Button variant="ghost" size="sm" className="h-7" title="Print tax invoice" onClick={() => void printInvoice(r)}><Printer className="h-3.5 w-3.5" /></Button>
   369→          <Button variant="ghost" size="sm" className="h-7 text-green-600 hover:text-green-700" title="Send on WhatsApp" onClick={() => void shareInvoice(r)}><MessageCircle className="h-3.5 w-3.5" /></Button>
   370→          <Button variant="ghost" size="sm" className="h-7" title="Email invoice" onClick={() => void emailInvoice(r)}><Mail className="h-3.5 w-3.5" /></Button>
   371→          {!r.irn && r.status !== 'CANCELLED' && (
   372→            <Button variant="ghost" size="sm" className="h-7" title="Generate IRN" onClick={() => void generateIrn(r)}><BadgeIndianRupee className="h-3.5 w-3.5" /></Button>
   373→          )}
   374→          {r.status !== 'PAID' && r.status !== 'CANCELLED' && (
   375→            <Button variant="outline" size="sm" className="h-7" onClick={() => void openPay(r)}>Payment</Button>
   376→          )}
   377→          {r.status !== 'CANCELLED' && (r.paidAmount || 0) <= 0 && (
   378→            <ConfirmAction
   379→              trigger={<Button variant="ghost" size="sm" className="h-7 text-red-600"><X className="h-3.5 w-3.5" /></Button>}
   380→              title="Cancel invoice?" message={`Cancel ${r.number}? Only unpaid invoices can be cancelled.`} tone="destructive"
   381→              onConfirm={() => cancelInvoice(r)}
   382→            />
   383→          )}
   384→        </div>
   385→      ),
   386→    },
   387→  ]
   388→
   389→  return (
   390→    <div>
   391→      <PageHeader
   392→        title="Sales Invoices"
   393→        description="GST tax invoices, payments received and e-invoicing"
   394→        actions={<Button onClick={() => { resetForm(); setOpen(true) }}><Plus className="h-4 w-4 mr-1.5" />New Invoice</Button>}
   395→      />
   396→      <DataTable columns={columns} rows={rows} loading={list.loading} searchPlaceholder="Search invoices…" highlightId={highlightId} moduleKey="sales-invoices" savedViewsKey={`${companyId}:sales-invoices`} />
   397→
   398→      <FormDialog
   399→        open={open} onOpenChange={setOpen} wide title="New Sales Invoice" submitLabel="Create Invoice"
   400→        description="From a sales order (pending qty pre-filled) or manual lines" loading={saving} onSubmit={submit}
   401→      >
   402→        <div className="grid sm:grid-cols-3 gap-4">
   403→          <Field label="Mode">
   404→            <SelectInput
   405→              value={mode} onChange={setMode}
   406→              options={[{ value: 'SO', label: 'From Sales Order' }, { value: 'MANUAL', label: 'Manual Lines' }]}
   407→            />
   408→          </Field>
   409→          {mode === 'SO' ? (
   410→            <Field label="Sales Order" required>
   411→              <SelectInput value={salesOrderId} onChange={(v) => void pickSo(v)} options={soOpts} placeholder="Select SO…" />
   412→            </Field>
   413→          ) : (
   414→            <Field label="Customer" required>
   415→              <SelectInput value={customerId} onChange={setCustomerId} options={customerOpts} />
   416→            </Field>
   417→          )}
   418→          <Field label="Date" required><DateInput value={date} onChange={setDate} /></Field>
   419→          <Field label="Due Date"><DateInput value={dueDate} onChange={setDueDate} /></Field>
   420→          <Field label="Place of Supply"><TextInput value={placeOfSupply} onChange={(e) => setPlaceOfSupply(e.target.value)} placeholder="e.g. Maharashtra" /></Field>
   421→          <Field label="Notes"><TextInput value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Remarks…" /></Field>
   422→        </div>
   423→
   424→        <div className="space-y-2">
   425→          <div className="flex items-center justify-between">
   426→            <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Lines</span>
   427→            <Button type="button" variant="outline" size="sm" onClick={() => setLines((ls) => [...ls, emptyLine()])}>
   428→              <Plus className="h-3.5 w-3.5 mr-1" />Add Line
   429→            </Button>
   430→          </div>
   431→          {lines.map((l, idx) => (
   432→            <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   433→              <div className="col-span-12 sm:col-span-4">
   434→                <SelectInput
   435→                  value={l.itemId} options={itemOpts} placeholder="Select item…"
   436→                  onChange={(v) => setLines((ls) => ls.map((x, i) => {
   437→                    if (i !== idx) return x
   438→                    const it = (items.data?.items ?? []).find((o) => o.id === v)
   439→                    return { ...x, itemId: v, rate: x.rate || Number(it?.sellingPrice || 0), gstRate: it?.gstRate ?? x.gstRate }
   440→                  }))}
   441→                />
   442→              </div>
   443→              <NumberInput className="col-span-3 sm:col-span-2" value={l.qty} min={0} placeholder="Qty"
   444→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, qty: n } : x)))} />
   445→              <NumberInput className="col-span-3 sm:col-span-2" value={l.rate} min={0} placeholder="Rate"
   446→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, rate: n } : x)))} />
   447→              <NumberInput className="col-span-3 sm:col-span-1" value={l.discountPct} min={0} placeholder="Disc %"
   448→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, discountPct: n } : x)))} />
   449→              <NumberInput className="col-span-2 sm:col-span-1" value={l.gstRate} min={0} placeholder="GST %"
   450→                onChange={(n) => setLines((ls) => ls.map((x, i) => (i === idx ? { ...x, gstRate: n } : x)))} />
   451→              <div className="col-span-1 flex justify-end">
   452→                <Button type="button" variant="ghost" size="sm" disabled={lines.length <= 1}
   453→                  onClick={() => setLines((ls) => ls.filter((_, i) => i !== idx))}>
   454→                  <Trash2 className="h-3.5 w-3.5 text-red-600" />
   455→                </Button>
   456→              </div>
   457→            </div>
   458→          ))}
   459→          <div className="rounded-md border bg-muted/30 p-3 text-sm grid grid-cols-2 sm:grid-cols-5 gap-2">
   460→            <TotCell label="Subtotal" value={totals.subtotal} />
   461→            <TotCell label="Discount" value={totals.discount} />
   462→            <TotCell label="CGST (preview)" value={totals.cgst} />
   463→            <TotCell label="SGST (preview)" value={totals.sgst} />
   464→            <TotCell label="Grand Total" value={totals.grand} bold />
   465→          </div>
   466→        </div>
   467→      </FormDialog>
   468→
   469→      <FormDialog
   470→        open={!!payFor} onOpenChange={(o) => !o && setPayFor(null)} title={`Record Payment — ${payFor?.number || ''}`}
   471→        submitLabel="Save Payment" loading={paying} onSubmit={submitPayment}
   472→        description={`Outstanding: ${fmtMoney(payFor ? (payFor.grandTotal || 0) - (payFor.paidAmount || 0) : 0)}`}
   473→      >
   474→        <div className="grid sm:grid-cols-2 gap-4">
   475→          <Field label="Amount" required><NumberInput value={payAmount} onChange={setPayAmount} min={0} /></Field>
   476→          <Field label="Mode">
   477→            <SelectInput value={payMode} onChange={setPayMode} options={PAYMENT_MODES.map((m) => ({ value: m, label: m }))} />
   478→          </Field>
   479→          <Field label="Date" required><DateInput value={payDate} onChange={setPayDate} /></Field>
   480→          <Field label="Reference"><TextInput value={payRef} onChange={(e) => setPayRef(e.target.value)} placeholder="UTR / cheque no." /></Field>
   481→        </div>
   482→      </FormDialog>
   483→    </div>
   484→  )
   485→}
   486→
   487→function TotCell({ label, value, bold }: { label: string; value: number; bold?: boolean }) {
   488→  return (
   489→    <div className={bold ? 'sm:text-right' : ''}>
   490→      <p className="text-[11px] text-muted-foreground uppercase tracking-wide">{label}</p>
   491→      <p className={`tabular-nums ${bold ? 'font-bold text-base' : 'font-medium'}`}>{fmtMoney(value)}</p>
   492→    </div>
   493→  )
   494→}
