    28→    28→
    29→    29→const r2 = (n: number) => Math.round(n * 100) / 100
    30→    30→const today = () => new Date().toISOString().slice(0, 10)

    33→    33→function computeTotals(lines: DraftLine[]) {
    34→    34→  let subtotal = 0, discount = 0, gst = 0
    35→    35→  for (const l of lines) {
    36→    36→    const gross = l.qty * l.rate
    37→    37→    const disc = gross * ((l.discountPct || 0) / 100)
    38→    38→    const net = gross - disc
    39→    39→    subtotal += gross; discount += disc; gst += net * ((l.gstRate || 0) / 100)
    40→    40→  }
    41→    41→  return { subtotal: r2(subtotal), discount: r2(discount), cgst: r2(gst / 2), sgst: r2(gst / 2), grand: r2(subtotal - discount + gst) }
    42→    42→}
    43→    43→function emptyLine(): DraftLine { return { itemId: '', qty: 1, rate: 0, discountPct: 0, gstRate: 18 } }

    46→    46→const esc = (v?: string | null) =>
    47→    47→  (v ?? '').replace(/[&<>"]/g, (c) => (c === '&' ? '&amp;' : c === '<' ? '&lt;' : c === '>' ? '&gt;' : '&quot;'))
    48→    48→// Join non-empty address parts with ', ' (raw — wrap the result in esc() before injecting).

    50→    50→  parts.filter((p): p is string => !!p && p.trim() !== '').join(', ')

    52→    52→// Build a wa.me deep link: 10-digit numbers get the '91' prefix, already-prefixed

    54→    54→function waUrl(phone: string | null | undefined, message: string): string {
    55→    55→  const digits = (phone || '').replace(/\D/g, '')
    56→    56→  const valid = digits.length === 10 || (digits.length === 12 && digits.startsWith('91'))
    57→    57→  return `https://wa.me/${valid ? digits : ''}?text=${encodeURIComponent(message)}`
    58→    58→}
